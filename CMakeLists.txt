cmake_minimum_required(VERSION 3.16)
project(PersonalFinanceManager VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(finance src/Finance.cpp)
target_include_directories(finance PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(finance PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
add_executable(finance_app src/main.cpp)
target_link_libraries(finance_app PRIVATE finance)
enable_testing()
add_executable(finance_tests tests/test_finance.cpp)
target_link_libraries(finance_tests PRIVATE finance)
add_test(NAME finance_tests COMMAND finance_tests)
